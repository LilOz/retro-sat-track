from pathlib import Path
import struct, collections, json, math
root=Path('orbit-cad-reve')
results=[]
for path in sorted(root.glob('*.stl')):
 data=path.read_bytes(); count=struct.unpack_from('<I',data,80)[0]
 assert len(data)==84+50*count, path.name
 edges=collections.Counter(); xyz=[]; volume=0
 for i in range(count):
  row=struct.unpack_from('<12fH',data,84+50*i)
  a,b,c=[tuple(round(v,5) for v in row[j:j+3]) for j in (3,6,9)]
  assert a!=b and b!=c and c!=a,path.name
  xyz.extend([a,b,c])
  for u,v in [(a,b),(b,c),(c,a)]:edges[tuple(sorted([u,v]))]+=1
  volume+=(a[0]*(b[1]*c[2]-b[2]*c[1])+a[1]*(b[2]*c[0]-b[0]*c[2])+a[2]*(b[0]*c[1]-b[1]*c[0]))/6
 bad=sum(n!=2 for n in edges.values());assert bad==0,(path.name,bad)
 dims=[round(max(p[j] for p in xyz)-min(p[j] for p in xyz),3) for j in range(3)]
 assert all(v<=256 for v in dims),dims
 results.append({'file':path.name,'triangles':count,'nonmanifold_edges':bad,'size_mm':dims,'volume_mm3':round(abs(volume),2)})
assert len(results)==7
(root/'mesh-validation.json').write_text(json.dumps(results,indent=2)+'\n',encoding='utf-8')
print(json.dumps(results))
